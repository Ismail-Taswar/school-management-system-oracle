fee_id_wise_report

select fm.fee_m_id, student_id||' - '||f_st_std_name(student_id) student, f_class_name(class_id) class, sessions, 
       to_char(collection_date, 'DD-Mon-RRRR') collection_date, nvl(paid_amount,0) paid, nvl(total_amount,0) total_amt,
       f_fee_amt_month(fee_amount_d_id) month, nvl(monthly_fee,0) monthly, nvl(BI_ANNUAL_FEE,0) bi_annual,
       nvl(ANNUAL_FEE,0) annual, nvl(total,0) total
from   fee_m fm, fee_d fd
where  fd.fee_m_id = fm.fee_m_id
order  by fm.fee_m_id
